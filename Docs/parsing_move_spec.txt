# a move specification containing

import re		# Makes re matching available to you

game_res_match := re.match(
			r"^(.*)(1-0|0-1|1/2-1/2)$",
			rem_str)

if game_res_match:
	rem_str, game_res = game_res_match.groups()

# Rest of the patterns follow.
# Processed in order.

# The programming code is
#   ommitted to concentrate
#   on the patterns
	
	^(.*)(\+|#)$		# two check/checkmate patterns
	^(.*)([+#])$		# [...bunch of single characters...]	

	^(.*?)(0-0|0-0-0)$			# king and queen castle
								# NOTE the .*? which is a lazy
								# instead of greedy grabbing
	
	(.*)(=[QRBN])$				# Promotion
	(.*)([a-h][1-8])$   		# destination square
	(.*)([x:])$					# capture
	
	([KQRBN])(.*)$				# Piece  could be a P for pawn
								#  Match indicates regular piece
								#  No match indicates pawn
	([a-h])(.*)$					# file disambigate
	([1-8])(.*)$					# rank disambigate
	
	Allow, but ignore leading/trailing white space
	^\s*(.*)(my_pattern)\s*$
